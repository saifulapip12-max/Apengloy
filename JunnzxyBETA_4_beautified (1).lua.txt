getgenv().PosHandler = function(remote, payload)
    if typeof(payload) ~= "buffer" then
        return
    end
    local payloadLen = buffer.len(payload)
    local opcode = payloadLen > 0 and buffer.readu8(payload, 0) or -1
    local seenKey = remote.Name .. "|op" .. opcode .. "|len" .. payloadLen
    State.seen[seenKey] = (State.seen[seenKey] or 0) + 1
    if opcode == 8 and payloadLen >= 9 then
        State.y = buffer.readf32(payload, 1)
        State.x = buffer.readf32(payload, 5)
    end
end

local function getPacketDiagnostics()
    local lines = {}
    for packetKey, seenCount in pairs(State.seen) do
        lines[#lines + 1] = packetKey .. " x" .. seenCount
    end
    table.sort(lines)
    if #lines == 0 then
        return "Belum ada paket ByteNet tertangkap sama sekali."
    end
    return table.concat(lines, "\n")
end

local function create(className, props, parent)
    local instance = Instance.new(className)
    pcall(function()
        instance.BorderSizePixel = 0
    end)
    for propName, propValue in pairs(props) do
        pcall(function()
            instance[propName] = propValue
        end)
    end
    instance.Parent = parent
    return instance
end
local function addCorner(cornerTarget, cornerRadius)
    pcall(function()
        local corner = Instance.new("UICorner")
        corner.CornerRadius = UDim.new(0, cornerRadius)
        corner.Parent = cornerTarget
    end)
end
local function addStroke(strokeTarget, strokeColor, strokeThickness, strokeTransparency)
    local stroke
    pcall(function()
        stroke = Instance.new("UIStroke")
        stroke.Color = strokeColor
        stroke.Thickness = strokeThickness or 1
        stroke.Transparency = strokeTransparency or 0
        stroke.ApplyStrokeMode = Enum.ApplyStrokeMode.Border
        stroke.Parent = strokeTarget
    end)
    return stroke
end
local function addGradient(gradientTarget, colorA, colorB, gradientRotation)
    pcall(function()
        local gradient = Instance.new("UIGradient")
        gradient.Color = ColorSequence.new(colorA, colorB)
        gradient.Rotation = gradientRotation or 0
        gradient.Parent = gradientTarget
    end)
end
local function addPadding(paddingTarget, padTop, padLeft, padRight, padBottom)
    create("UIPadding", {
        PaddingTop = UDim.new(0, padTop or 0),
        PaddingLeft = UDim.new(0, padLeft or 0),
        PaddingRight = UDim.new(0, padRight or 0),
        PaddingBottom = UDim.new(0, padBottom or 0),
    }, paddingTarget)
end
local function tween(tweenTarget, tweenDuration, tweenGoal, easingStyle, easingDirection)
    local tweenOk, tweenObj = pcall(function()
        return TweenService:Create(tweenTarget, TweenInfo.new(tweenDuration, easingStyle or Enum.EasingStyle.Quad, easingDirection or Enum.EasingDirection.Out), tweenGoal)
    end)
    if tweenOk and tweenObj then
        tweenObj:Play()
        return tweenObj
    end
end
local function addHoverStroke(hoverButton)
    local hoverStroke = addStroke(hoverButton, Theme.acc2, 1, 1)
    if not hoverStroke then

    hoverButton.MouseEnter:Connect(function()

    end)
    hoverButton.MouseLeave:Connect(function()
        tween(hoverStroke, 0.15, {Transparency = 1})
    end)
end

local function selectTab(tabName)
    if currentTab == tabName then
        return
    end
    currentTab = tabName
    for otherTabName, tabData in pairs(Tabs) do
        local isSelected = otherTabName == tabName
        tween(tabData.btn, 0.18, {BackgroundTransparency = isSelected and 0.35 or 1})
        tween(tabData.ind, 0.18, {BackgroundTransparency = isSelected and 0 or 1})
        tween(tabData.tl, 0.18, {TextColor3 = isSelected and Theme.wh or Theme.dm})
        tabData.page.Visible = isSelected
        if isSelected then
            PageTitle.Text = otherTabName
            PageDesc.Text = tabData.desc or ""
            tabData.page.Position = UDim2.new(0, 14, 0, 0)
            tween(tabData.page, 0.22, {Position = UDim2.new(0, 0, 0, 0)})
        end
    end
end

local function makeDraggable(dragHandle, dragTarget, onClick)
    local dragging, dragStart, startPos, dragMoved = false, nil, nil, false
    dragHandle.InputBegan:Connect(function(inputBegin)
        if inputBegin.UserInputType == Enum.UserInputType.MouseButton1 or inputBegin.UserInputType == Enum.UserInputType.Touch then
            dragging, dragMoved = true, false
            dragStart, startPos = inputBegin.Position, dragTarget.Position
            inputBegin.Changed:Connect(function()
                if inputBegin.UserInputState == Enum.UserInputState.End then
                    if dragging and not dragMoved and onClick then
                        onClick()
                    end
                    dragging = false
                end
            end)
        end
    end)
    Connections[#Connections + 1] = UserInputService.InputChanged:Connect(function(inputMove)
        if dragging and (inputMove.UserInputType == Enum.UserInputType.MouseMovement or inputMove.UserInputType == Enum.UserInputType.Touch) then
            local dragDelta = inputMove.Position - dragStart
            if dragDelta.Magnitude > 4 then
                dragMoved = true
            end
            dragTarget.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + dragDelta.X, startPos.Y.Scale, startPos.Y.Offset + dragDelta.Y)
        end
    end)
end
local cachedMoveToRemote = nil
local function getRemotes()
    if not (cachedMoveToRemote and cachedMoveToRemote.Parent) then
        cachedMoveToRemote = nil
        for _, rsDescendant in ipairs(ReplicatedStorage:GetDescendants()) do
            if rsDescendant.Name == "MoveTo" and rsDescendant:IsA("RemoteEvent") then
                cachedMoveToRemote = rsDescendant
                break
            end
        end
    end
    return ReplicatedStorage:FindFirstChild("ByteNetReliable"), cachedMoveToRemote
end
local function buildActionPacket(actionId, packetX, packetY)
    local actionBuf = buffer.create(10)
    buffer.writeu8(actionBuf, 0, 3)
    buffer.writef32(actionBuf, 1, packetY)
    buffer.writeu8(actionBuf, 5, actionId)
    buffer.writef32(actionBuf, 6, packetX)
    return actionBuf
end
local function buildPlacePacket(placeTileX, placeTileY, placeBlockId)
    local placeBuf = buffer.create(9)
    buffer.writeu8(placeBuf, 0, 32)
    buffer.writeu16(placeBuf, 1, placeTileY)
    buffer.writeu16(placeBuf, 3, placeTileX)
    buffer.writeu8(placeBuf, 5, 0)
    buffer.writeu16(placeBuf, 6, placeBlockId)
    buffer.writeu8(placeBuf, 8, 1)
    return placeBuf
end
local cachedInventory, cachedInventoryUI = nil, nil
local function findInventory(forceRefresh)
    if not forceRefresh and cachedInventory and type(rawget(cachedInventory, "slots")) == "table" then
        return cachedInventory, cachedInventoryUI
    end
    cachedInventory, cachedInventoryUI = nil, nil
    if not getgc then
        return nil, nil
    end
    local gcOk, gcList = pcall(getgc, true)
    if not gcOk or type(gcList) ~= "table" then
        return nil, nil
    end
    for _, gcObj in ipairs(gcList) do
        if type(gcObj) == "table" then
            if rawget(gcObj, "capacity") and type(rawget(gcObj, "slots")) == "table" then
                local invSlots = rawget(gcObj, "slots")
                if rawget(gcObj, "selected") ~= nil or rawget(invSlots, 1) ~= nil then
                    cachedInventory = gcObj
                end
            end
            if not cachedInventoryUI and type(rawget(gcObj, "refresh")) == "function" and rawget(gcObj, "screen") ~= nil then
                cachedInventoryUI = gcObj
            end
        end
    end
    return cachedInventory, cachedInventoryUI
end
local function getItemCount(countItemId, refreshInv)
    local inventory = findInventory(refreshInv)
    if not inventory then
        inventory = findInventory(true)
    end
    if not inventory then
        return nil
    end
    local slots = rawget(inventory, "slots")
    if type(slots) ~= "table" then
        return nil
    end
    local total = 0
    for _, slot in pairs(slots) do
        if type(slot) == "table" and tonumber(rawget(slot, "id")) == countItemId then
            total = total + (tonumber(rawget(slot, "n")) or 0)
        end
    end
    return total
end
local function consumeItem(consumeId, consumeAmount)
    consumeAmount = tonumber(consumeAmount) or 1
    if consumeAmount <= 0 then
        return
    end
    pcall(function()
        local inv, invUI = findInventory(false)
        if not inv then
            inv, invUI = findInventory(true)
        end
        if not inv then
            return
        end
        local invSlotList = rawget(inv, "slots")
        if type(invSlotList) ~= "table" then
            return
        end
        local remaining = consumeAmount
        for slotIndex, invSlot in pairs(invSlotList) do
            if type(invSlot) == "table" and tonumber(rawget(invSlot, "id")) == consumeId then
                local slotCount = tonumber(rawget(invSlot, "n")) or 0
                if slotCount > remaining then
                    rawset(invSlot, "n", slotCount - remaining)
                    remaining = 0
                    break
                else
                    remaining = remaining - slotCount
                    invSlotList[slotIndex] = nil
                    if remaining <= 0 then
                        break
                    end
                end
            end
        end
        if invUI and type(rawget(invUI, "refresh")) == "function" then
            task.spawn(rawget(invUI, "refresh"))
        end
    end)
end
local setTileFn, dropListFn, worldState = nil, nil, nil
local function findGameFunctions(forceScan)
    if not forceScan and setTileFn then
        return setTileFn
    end
    pcall(function()
        local worldRemotes = ReplicatedStorage:FindFirstChild("WorldRemotes")
        local tileChanged = worldRemotes and worldRemotes:FindFirstChild("TileChanged")
        if tileChanged and getconnections then
            for _, connection in ipairs(getconnections(tileChanged.OnClientEvent)) do
                if connection.Function then
                    local upOk, upvalues = pcall(debug.getupvalues, connection.Function)
                    if upOk and type(upvalues) == "table" and type(upvalues[2]) == "function" then
                        setTileFn = upvalues[2]
                    end
                end
            end
        end
    end)
    if getgc then
        pcall(function()
            for _, gcItem in ipairs(getgc(true)) do
                if type(gcItem) == "function" then
                    local infoOk, sourceName, funcName = pcall(function()
                        local infoSource, _, infoName = debug.info(gcItem, "sln")
                        return infoSource, tostring(infoName or "")
                    end)
                    if infoOk and sourceName and tostring(sourceName):find("GameGui.Game") then
                        if funcName == "setTile" and not setTileFn then
                            setTileFn = gcItem
                        elseif funcName == "dropList" then
                            dropListFn = gcItem
                        elseif funcName == "hit" or funcName == "act" then
                            local hitUpOk, hitUpvalues = pcall(debug.getupvalues, gcItem)
                            if hitUpOk and type(hitUpvalues) == "table" then
                                for _, upvalue in pairs(hitUpvalues) do
                                    if type(upvalue) == "table" and type(rawget(upvalue, "fg")) == "table" and type(rawget(upvalue, "w")) == "number" then
                                        worldState = upvalue
                                    end
                                end
                            end
                        end
                    end
                elseif type(gcItem) == "table" then
                    if not worldState and type(rawget(gcItem, "fg")) == "table" and type(rawget(gcItem, "w")) == "number" then
                        worldState = gcItem
                    end
                end
            end
        end)
    end
    return setTileFn
end
local function getTileId(queryX, queryY)
    if not worldState then
        return nil
    end
    local fgGrid, worldWidth = rawget(worldState, "fg"), rawget(worldState, "w")
    if type(fgGrid) == "table" and type(worldWidth) == "number" and worldWidth > 0 then
        local cell = fgGrid[queryY * worldWidth + queryX + 1]
        if type(cell) == "number" then
            return cell
        end
        if type(cell) == "table" then
            return tonumber(rawget(cell, "id")) or 0
        end
    end
    return nil
end
local function callSetTile(layer, setX, setY, setId, setExtra)
    local setTile = findGameFunctions(true)
    if setTile then
        if setExtra ~= nil then
            pcall(setTile, layer, setX, setY, setId, setExtra)
        else
            pcall(setTile, layer, setX, setY, setId)
        end
        return true
    end
    return false
end
local cachedDropList = nil
local function isDropCollected(drop)
    if rawget(drop, "pending") or rawget(drop, "picked") then
        return true
    end
    if rawget(drop, "from") ~= nil then
        return true
    end
    local dropFrame = rawget(drop, "frame")
    if dropFrame ~= nil and typeof(dropFrame) == "Instance" and dropFrame.Parent == nil then
        return true
    end
    return false
end
local function isDropEntry(entry)
    if type(entry) ~= "table" or isDropCollected(entry) then
        return false
    end
    local entryX = rawget(entry, "x") or rawget(entry, "X") or rawget(entry, "px")
    local entryY = rawget(entry, "y") or rawget(entry, "Y") or rawget(entry, "py")
    if type(entryX) == "number" and type(entryY) == "number" and entryX >= 0 and entryX <= 200 and entryY >= 0 and entryY <= 200 then
        if rawget(entry, "id") ~= nil or rawget(entry, "item") ~= nil or rawget(entry, "itemId") ~= nil or rawget(entry, "count") ~= nil or rawget(entry, "n") ~= nil or rawget(entry, "uid") ~= nil then
            if rawget(entry, "capacity") == nil and rawget(entry, "slots") == nil and rawget(entry, "fg") == nil then
                return true
            end
        end
    end
    local entryPos = rawget(entry, "pos") or rawget(entry, "position") or rawget(entry, "p")
    local entryPosType = typeof(entryPos)
    if (entryPosType == "Vector2" or entryPosType == "Vector3") and entryPos.X >= 0 and entryPos.X <= 200 and entryPos.Y >= 0 and entryPos.Y <= 200 then
        if rawget(entry, "id") ~= nil or rawget(entry, "item") ~= nil or rawget(entry, "count") ~= nil or rawget(entry, "n") ~= nil then
            return true
        end
    end
    return false
end
local function getDropPosition(dropEntry)
    local dropX = rawget(dropEntry, "x") or rawget(dropEntry, "X") or rawget(dropEntry, "px")
    local dropY = rawget(dropEntry, "y") or rawget(dropEntry, "Y") or rawget(dropEntry, "py")
    if type(dropX) == "number" and type(dropY) == "number" then
        return dropX, dropY
    end
    local dropPos = rawget(dropEntry, "pos") or rawget(dropEntry, "position") or rawget(dropEntry, "p")
    local dropPosType = typeof(dropPos)
    if dropPosType == "Vector2" or dropPosType == "Vector3" then
        return dropPos.X, dropPos.Y
    end
    return nil, nil
end
local function findDropsNear(centerX, centerY, searchRadius)
    searchRadius = searchRadius or 4.5
    local found, visited = {}, {}
    local function scanTable(tbl)
        if type(tbl) ~= "table" then
            return
        end
        for _, candidate in pairs(tbl) do
            if type(candidate) == "table" and not visited[candidate] and isDropEntry(candidate) then
                local candX, candY = getDropPosition(candidate)
                if candX and candY and math.abs(candX - (centerX + 0.5)) <= searchRadius and math.abs(candY - (centerY + 0.5)) <= searchRadius then
                    visited[candidate] = true
                    found[#found + 1] = {
                        raw = candidate,
                        x = candX,
                        y = candY,
                        cx = math.floor(candX),
                        cy = math.floor(candY - 0.001),
                    }
                end
            end
        end
    end
    findGameFunctions(false)
    if dropListFn then
        local dlOk, dlResult = pcall(dropListFn)
        if dlOk and type(dlResult) == "table" then
            scanTable(dlResult)
            return found
        else
            local dlUpOk, dlUpvalues = pcall(debug.getupvalues, dropListFn)
            if dlUpOk and type(dlUpvalues) == "table" then
                for _, dlUpvalue in pairs(dlUpvalues) do
                    if type(dlUpvalue) == "table" then
                        scanTable(dlUpvalue)
                        if #found > 0 then
                            return found
                        end
                    end
                end
            end
        end
    end
    if #found == 0 and cachedDropList then
        scanTable(cachedDropList)
    end
    if #found == 0 and getgc then
        local gcOk2, gcList2 = pcall(getgc, true)
        if gcOk2 and type(gcList2) == "table" then
            for _, gcTable in ipairs(gcList2) do
                if type(gcTable) == "table" then
                    local dropTable = rawget(gcTable, "drops") or rawget(gcTable, "dropList")
                    if type(dropTable) == "table" then
                        local beforeCount = #found
                        scanTable(dropTable)
                        if #found > beforeCount then
                            cachedDropList = dropTable
                        end
                    end
                end
            end
        end
    end
    return found
end
local heldKey = nil
local function sendKey(keyCode, isPressed)
    pcall(function()
        VirtualInputManager:SendKeyEvent(isPressed, keyCode, false, game)
    end)
end
local function releaseMoveKeys()
    heldKey = nil
    sendKey(Enum.KeyCode.A, false)
    sendKey(Enum.KeyCode.D, false)
end
local function holdMoveKey(moveKey)
    if heldKey ~= moveKey then
        releaseMoveKeys()
        sendKey(moveKey, true)
        heldKey = moveKey
    end
end
local function jump()
    sendKey(Enum.KeyCode.Space, true)
    task.wait(0.12)
    sendKey(Enum.KeyCode.Space, false)
end
local function toCell(posX, posY)
    return math.floor(posX), math.floor(posY - 0.001)
end
local function teleport(tpX, tpY)
    local _, moveToRemote = getRemotes()
    if moveToRemote then
        pcall(firesignal, moveToRemote.OnClientEvent, tpX, tpY)
    end
    State.x, State.y = tpX, tpY
end
local function moveToPoint(targetX, targetY, waitItems, maxWait, shouldStop)
    releaseMoveKeys()
    maxWait = maxWait or 0.18
    teleport(targetX, targetY)
    local startTime = os.clock()
    local lastTp = startTime
    while not (shouldStop and shouldStop()) and os.clock() - startTime < maxWait do
        if type(waitItems) == "table" and #waitItems > 0 then
            local allCollected, pendingItem = true, nil
            for _, waitItem in ipairs(waitItems) do
                if not isDropCollected(waitItem) then
                    allCollected = false
                    pendingItem = waitItem
                    break
                end
            end
            if allCollected then
                break
            end
            if os.clock() - lastTp >= 0.05 then
                lastTp = os.clock()
                local itemX, itemY = getDropPosition(pendingItem)
                if itemX and itemY then
                    teleport(itemX, itemY)
                else
                    teleport(targetX, targetY)
                end
            end
        elseif os.clock() - lastTp >= 0.06 then
            lastTp = os.clock()
            teleport(targetX, targetY)
        end
        task.wait(0.03)
    end
end
local function teleportToCell(cellX, cellY, cellWait, cellStop)
    moveToPoint(cellX + 0.5, cellY + 0.999, nil, cellWait, cellStop)
end
local function snapToCell(snapX, snapY, snapStop)
    releaseMoveKeys()
    for _ = 1, 3 do
        if snapStop and snapStop() then
            break
        end
        teleport(snapX + 0.5, snapY + 0.999)
        task.wait(0.06)
    end
end
local function walkToCell(goalX, goalY, walkStop, walkTimeout)
    walkTimeout = walkTimeout or 35
    local walkStart = os.clock()
    local lastX, lastY, lastMoveTime = State.x or 0, State.y or 0, os.clock()
    while not (walkStop and walkStop()) and os.clock() - walkStart < walkTimeout do
        if not (State.x and State.y) then
            task.wait(0.05)
            continue
        end
        local curCellX, curCellY = toCell(State.x, State.y)
        local diffX = goalX + 0.5 - State.x
        if curCellX == goalX and curCellY == goalY and math.abs(diffX) < 0.28 then
            releaseMoveKeys()
            task.wait(0.08)
            local checkX, checkY = toCell(State.x, State.y)
            if checkX == goalX and checkY == goalY then
                return true
            end
        end
        if math.abs(diffX) > 0.12 then
            holdMoveKey(diffX > 0 and Enum.KeyCode.D or Enum.KeyCode.A)
        else
            releaseMoveKeys()
        end
        if curCellY > goalY and math.abs(diffX) < 1.5 then
            jump()
        end
        task.wait(0.03)
        if math.abs(State.x - lastX) + math.abs(State.y - lastY) > 0.03 then
            lastX, lastY, lastMoveTime = State.x, State.y, os.clock()
        elseif os.clock() - lastMoveTime > 0.55 then
            jump()
            lastMoveTime = os.clock()
        end
    end
    releaseMoveKeys()
    if not (State.x and State.y) then
        return false
    end
    local endX, endY = toCell(State.x, State.y)
    return endX == goalX and endY == goalY
end

    local function collectDrops(meX, meY, brokenTiles, collectDelay, verifyMax, collectStop)
        local collectRadius = 4.5
        findGameFunctions(false)
        task.wait(0.22)
        local dropGroups = {}
        for _, nearDrop in ipairs(findDropsNear(meX, meY, collectRadius)) do
            local groupKey = nearDrop.cx .. "," .. nearDrop.cy
            if not dropGroups[groupKey] then
                dropGroups[groupKey] = {x = nearDrop.x, y = nearDrop.y, items = {nearDrop.raw}}
            else
                local group = dropGroups[groupKey]
                group.items[#group.items + 1] = nearDrop.raw
            end
        end

            if collectStop() then
                break

            local tileKey = clampX .. "," .. clampY
            visitedKeys[tileKey] = true
            local tileGroup = dropGroups[tileKey]
            setStatus(string.format("Collect Tile #%d -> (%d, %d)", brokenTile.num, clampX, clampY), Theme.ON)
            if tileGroup then
                moveToPoint(tileGroup.x, tileGroup.y, tileGroup.items, collectDelay, collectStop)
            else
                teleportToCell(clampX, clampY, collectDelay, collectStop)
            end
        end
        for leftKey, leftGroup in pairs(dropGroups) do
            if collectStop() then
                break
            end
            if not visitedKeys[leftKey] then
                moveToPoint(leftGroup.x, leftGroup.y, leftGroup.items, collectDelay, collectStop)
            end
        end
        for verifyRound = 1, verifyMax do
            if collectStop() then
                break
            end
            task.wait(0.12)
            local remainingDrops = findDropsNear(meX, meY, collectRadius)
            if #remainingDrops == 0 then
                break
            end
            setStatus(string.format("Verifikasi Sisa Drop #%d (%d item)...", verifyRound, #remainingDrops), Theme.ON)
            for _, leftover in ipairs(remainingDrops) do
                if collectStop() then
                    break
                end
                if not isDropCollected(leftover.raw) then
                    moveToPoint(leftover.x, leftover.y, {leftover.raw}, collectDelay, collectStop)
                end
            end
        end
        if not collectStop() then

        end
    end

            local suOk, suUpvalues = pcall(debug.getupvalues, scanFn)

                return

            for _, suValue in pairs(suUpvalues) do

                            return conn.Function

                        end

        local fgWorld = getWorld()

    local function isBlocked(blkX, blkY)
        local blkWorld = getWorld()
        if not blkWorld or blkX < 0 or blkX >= blkWorld.w or blkY < 0 or blkY >= blkWorld.h then
            return true
        end
        local blkId = getFgId(blkX, blkY)
        if blkId == 0 then
            return false
        end
        local blkItem = blkWorld.items[blkId]
        if not blkItem then
            return true
        end
        return blkItem.solid == true or blkItem.hurts == true
    end
    local function findPath(fromX, fromY, toX, toY)
        local function packKey(pkX, pkY)
            return pkY * 256 + pkX
        end
        local cameFrom = {[packKey(fromX, fromY)] = false}
        local queue, head = {{fromX, fromY}}, 1
        local directions = {{1, 0}, {-1, 0}, {0, 1}, {0, -1}}
        while head <= #queue and head < 6000 do
            local node = queue[head]
            head = head + 1
            if node[1] == toX and node[2] == toY then
                local path, step = {}, node
                while step do
                    table.insert(path, 1, step)
                    step = cameFrom[packKey(step[1], step[2])]
                end
                return path
            end
            for _, dir in ipairs(directions) do
                local nx, ny = node[1] + dir[1], node[2] + dir[2]
                local nkey = packKey(nx, ny)
                if cameFrom[nkey] == nil and not isBlocked(nx, ny) then
                    cameFrom[nkey] = node
                    queue[#queue + 1] = {nx, ny}
                end
            end
        end
    end

        if type(wdList) ~= "table" then
            return nil

        for _, wdEntry in pairs(wdList) do
            if type(wdEntry) == "table" then
                local wdX = rawget(wdEntry, "x") or rawget(wdEntry, "X") or rawget(wdEntry, "px")
                local wdY = rawget(wdEntry, "y") or rawget(wdEntry, "Y") or rawget(wdEntry, "py")
                if not (type(wdX) == "number" and type(wdY) == "number") then
                    wdX, wdY = nil, nil
                    local wdPos = rawget(wdEntry, "pos") or rawget(wdEntry, "position") or rawget(wdEntry, "p")
                    local wdPosType = typeof(wdPos)
                    if wdPosType == "Vector2" or wdPosType == "Vector3" then
                        wdX, wdY = wdPos.X, wdPos.Y
                    end
                end
                if type(wdX) == "number" and type(wdY) == "number" and wdX >= 0 and wdX <= 100 and wdY >= 0 and wdY <= 60 then
                    wdResult[#wdResult + 1] = {x = wdX, y = wdY}
                end
            end
        end
        return wdResult
    end

        end

local FloatingButton = create("TextButton", {
    Size = UDim2.new(0, 46, 0, 46),
    Position = UDim2.new(0, 14, 0.5, -23),
    AutoButtonColor = false,
    BackgroundColor3 = Theme.wh,
    Text = "J",
    TextColor3 = Theme.wh,
    TextSize = 19,
    Font = Fonts.b,
    Visible = false,
}, ScreenGui)
addCorner(FloatingButton, 23)
addGradient(FloatingButton, Theme.acc, Theme.acc2, 45)
addStroke(FloatingButton, Theme.wh, 1.5, 0.3)
do
    local function getGlassTransparency(bgColor)
        if bgColor == Theme.bg then
            return 0.25
        end
        if bgColor == Theme.panel then
            return 0.38
        end
        if bgColor == Theme.card then
            return 0.52
        end
        if bgColor == Theme.card2 then
            return 0.45
        end
    end
    for _, guiObj in ipairs(ScreenGui:GetDescendants()) do
        if guiObj:IsA("GuiObject") and guiObj ~= MainFrame and guiObj.BackgroundTransparency < 1 then
            if guiObj.Parent == TopBar and guiObj.ClassName == "Frame" then
                guiObj.BackgroundTransparency = 1
            else
                local glassAlpha = getGlassTransparency(guiObj.BackgroundColor3)
                if glassAlpha then
                    guiObj.BackgroundTransparency = glassAlpha
                end
            end
        end
    end
    MainFrame.BackgroundColor3 = Color3.fromRGB(255, 255, 255)
    MainFrame.BackgroundTransparency = 0.2
    pcall(function()
        local mainGradient = Instance.new("UIGradient")
        mainGradient.Color = ColorSequence.new(Color3.fromRGB(46, 40, 96), Color3.fromRGB(12, 12, 24))
        mainGradient.Rotation = 55
        mainGradient.Parent = MainFrame
    end)
    local mainStroke = MainFrame:FindFirstChildOfClass("UIStroke")
    if mainStroke then
        mainStroke.Color = Theme.wh
        mainStroke.Transparency = 0.72
        mainStroke.Thickness = 1.2
    end
    task.spawn(function()
        local logoImage, logoError = getLogoImage()
        if not logoImage then
            if logoError then
                notify("Logo: " .. logoError, Theme.YEL)
            end
            return
        end
        local function applyLogo(logoParent, logoRadius)
            local logoImg = Instance.new("ImageLabel")
            logoImg.Size = UDim2.new(1, 0, 1, 0)
            logoImg.BackgroundTransparency = 1
            logoImg.Image = logoImage
            logoImg.ScaleType = Enum.ScaleType.Crop
            logoImg.ZIndex = 5
            logoImg.Parent = logoParent
            addCorner(logoImg, logoRadius)
        end
        LogoLabel.Text = ""
        applyLogo(LogoLabel, 9)
        FloatingButton.Text = ""
        applyLogo(FloatingButton, 23)
    end)
end
local isAnimating = false
local function showMain()
    if isAnimating then
        return
    end
    MainFrame.Visible = true
    FloatingButton.Visible = false
    MainScale.Scale = UI_SCALE * 0.85
    tween(MainScale, 0.28, {Scale = UI_SCALE}, Enum.EasingStyle.Back)
end
local function hideMain()
    if isAnimating or not MainFrame.Visible then
        return
    end
    isAnimating = true
    tween(MainScale, 0.16, {Scale = UI_SCALE * 0.85})
    task.delay(0.16, function()
        MainFrame.Visible = false
        FloatingButton.Visible = true
        isAnimating = false
    end)
end
makeDraggable(TopBar, MainFrame)
makeDraggable(FloatingButton, FloatingButton, showMain)
MinimizeButton.Activated:Connect(hideMain)
function closeHub()
    stopAll(true)
    releaseMoveKeys()
    for _, hubConn in ipairs(Connections) do
        pcall(function()
            hubConn:Disconnect()
        end)
    end
    tween(MainScale, 0.18, {Scale = UI_SCALE * 0.8})
    task.delay(0.2, function()
        pcall(function()
            ScreenGui:Destroy()
        end)
    end)
end
getgenv().HubStop = closeHub
CloseButton.Activated:Connect(closeHub)
Connections[#Connections + 1] = UserInputService.InputBegan:Connect(function(keyInput, gameProcessed)
    if gameProcessed then
        return
    end
    if keyInput.KeyCode == Enum.KeyCode.RightShift then
        if MainFrame.Visible then
            hideMain()
        else
            showMain()
        end
    end
end)
